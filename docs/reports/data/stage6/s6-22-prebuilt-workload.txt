Команда из корня репозитория. RAW и исходник доступны контейнеру только для
чтения. Образ rust:1.99.0-bookworm для linux/amd64 был заранее доступен в
локальном OrbStack. GCC 15.2.0-16ubuntu1 и libc6 2.43-2ubuntu2.3 берутся из RAW.
Новые выходные файлы хранятся в work.

    set -eu
    raw="$PWD/work/stage5/s5-21/stage5-rootfs.raw"
    source_file="$PWD/guest/linux/tests/s6_smp_workloads.c"
    image=sha256:19059486ff9ddb4c6895cbc02a186f00274c257adcc6f79611ca29f55d36cdda
    test "$(shasum -a 256 "$raw" | awk '{print $1}')" = c2b2c94665ea256c6fa19c7eeebfc8451589028df3da00b2e1e05c7473dd2895
    mkdir -p "$PWD/work/stage6/s6-22-rootfs"
    workdir=$(mktemp -d "$PWD/work/stage6/s6-22-rootfs/prebuilt.XXXXXX")
    docker run --rm --pull=never --network=none --read-only --platform linux/amd64 \
      --memory=6g --tmpfs /var/tmp:size=4g,exec \
      --mount "type=bind,src=$raw,dst=/input.raw,readonly" \
      --mount "type=bind,src=$source_file,dst=/s6_smp_workloads.c,readonly" \
      "$image" bash -ec '
        root=/var/tmp/rootfs
        mkdir "$root"
        debugfs -R "rdump / $root" /input.raw >&2
        install -m 0644 /s6_smp_workloads.c "$root/tmp/s6_smp_workloads.c"
        chroot "$root" /usr/bin/dpkg-query -W libc6 >&2
        chroot "$root" /usr/bin/gcc-15 --version | grep -F "15.2.0-16ubuntu1" >&2
        for generation in a b; do
          chroot "$root" /usr/bin/env LC_ALL=C SOURCE_DATE_EPOCH=0 TMPDIR=/tmp \
            /usr/bin/gcc-15 -std=c11 -O2 -Wall -Wextra -Wpedantic -Werror -pthread \
            /tmp/s6_smp_workloads.c -o "/tmp/s6_smp_workloads.$generation"
        done
        cmp "$root/tmp/s6_smp_workloads.a" "$root/tmp/s6_smp_workloads.b"
        cat "$root/tmp/s6_smp_workloads.a"
      ' > "$workdir/workloads.elf"
    python3 tools/build/build_stage6_smp_rootfs_initramfs.py \
      --stage5-initramfs work/stage5/s5-09-inputs/stage5-initramfs.cpio.gz \
      --prebuilt-workload "$workdir/workloads.elf" --out "$workdir/initramfs.cpio.gz"

После распаковки префикс initramfs S5 совпал побайтно. Исходник, ELF и скрипт
из хвоста архива были извлечены в рабочую копию rootfs. GCC в этой копии
переименован перед запуском скрипта. Список online прочитан из sysfs
локального контейнера и скопирован в chroot для сценария. Привязка нагрузки
к CPU 2 и 5 проверялась гостевой программой через getcpu.
